pragma solidity ^0.8.20;

contract StudentData{
    struct Student{
        uint rollNo;
        string name;
        string department;
        uint marks;
    }

    Student[] public students;

    function addStudent(uint r, string memory n, string memory d, uint m) public{
        students.push(Student(r, n, d, m));
    }

    function getStudent(uint i) public view returns(uint, string memory, string memory , uint){
        require(i < students.length, "Invalid index");
        Student memory s = students[i];
        return (s.rollNo, s.name, s.department, s.marks);
    }

    function getStudentCount() public view returns (uint){
        return students.length;
    }
}
